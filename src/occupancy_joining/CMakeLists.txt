cmake_minimum_required(VERSION 3.16)
project(occupancy_joining VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(yaml-cpp REQUIRED)
add_library(joining_core src/core.cpp src/observed_inverse.cpp src/globalized_inverse.cpp src/block_schur.cpp)
target_include_directories(joining_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(joining_core PUBLIC Eigen3::Eigen yaml-cpp)
add_executable(joining_backend src/main.cpp)
target_link_libraries(joining_backend PRIVATE joining_core)
install(TARGETS joining_core joining_backend EXPORT joiningTargets ARCHIVE DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY scripts/ DESTINATION lib/${PROJECT_NAME} USE_SOURCE_PERMISSIONS PATTERN "__pycache__" EXCLUDE)
find_package(ament_cmake REQUIRED)
find_package(slam_toolbox QUIET)
if(slam_toolbox_FOUND)
  find_package(rclcpp_lifecycle REQUIRED)
  find_package(Boost REQUIRED COMPONENTS serialization)
  find_library(JOINING_KARTO_LIBRARY NAMES kartoSlamToolbox HINTS "${slam_toolbox_DIR}/../../.." PATH_SUFFIXES lib REQUIRED)
  add_executable(joining_export_slam_graph src/export_slam_graph.cpp)
  ament_target_dependencies(joining_export_slam_graph rclcpp_lifecycle slam_toolbox)
  target_link_libraries(joining_export_slam_graph Boost::serialization ${JOINING_KARTO_LIBRARY})
  install(TARGETS joining_export_slam_graph RUNTIME DESTINATION lib/${PROJECT_NAME})
else()
  message(STATUS "Final graph exporter disabled: slam_toolbox development package unavailable")
endif()

find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)
add_library(joining_grayscale_map SHARED src/grayscale_map.cpp)
ament_target_dependencies(joining_grayscale_map rviz_default_plugins pluginlib)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS joining_grayscale_map LIBRARY DESTINATION lib)
install(EXPORT joiningTargets DESTINATION share/${PROJECT_NAME}/cmake)
ament_export_targets(joiningTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 yaml-cpp)
ament_package()
